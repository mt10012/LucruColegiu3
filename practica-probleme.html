#include <iostream>
using namespace std;

const int DIMENSIUNE = 27;

class CovorulSierpinski {
private:
    char matrice[DIMENSIUNE][DIMENSIUNE];

public:
    CovorulSierpinski() {
        initializeaza();
    }

    void initializeaza() {
        for (int i = 0; i < DIMENSIUNE; ++i) {
            for (int j = 0; j < DIMENSIUNE; ++j) {
                matrice[i][j] = '#';
            }
        }
    }

    void genereazaPrimesIteratie(int startLinie, int startColoana, int marime) {
        if (marime < 1) return;

        int dimensiuneCelula = marime / 3;
        for (int i = startLinie + dimensiuneCelula; i < startLinie + 2 * dimensiuneCelula; ++i) {

            for (int j = startColoana + dimensiuneCelula; j < startColoana + 2 * dimensiuneCelula; ++j) {
                matrice[i][j] = ' ';
            }
        }
    }

    void afiseaza() {
        cout << "Covorul lui Sierpinski \n";
        for (int i = 0; i < DIMENSIUNE; ++i) {
            for (int j = 0; j < DIMENSIUNE; ++j) {
                cout << matrice[i][j] << matrice[i][j];
            }
            cout << "\n";
        }

        cout << "\n\nInformatii:\n";
        cout << "- Patratul initial este impartit in 9 parti egale (3x3)\n";
        cout << "- Patratul din mijloc este eliminat\n";
        cout << "- Patratelele ramase: 8\n";
        cout << "- Procent acoperire: " << (100.0 * 8 / 9) << "%\n";
    }
};

int main() {
    CovorulSierpinski covor;
    covor.genereazaPrimesIteratie(0, 0, DIMENSIUNE);
    covor.afiseaza();

    return 0;
}

  class Program
  {
    static int DigitSum(int number)
    {
      int sum = 0;
      while (number > 0)
      {
        sum += number % 10;
        number /= 10;
      }
      return sum;
    }

    static void Main()
    {
      int n = 0;
      try
      {
        if (File.Exists("input.txt"))
        {
          string content = File.ReadAllText("input.txt").Trim();
          n = int.Parse(content);
        }
        else
        {
          Console.WriteLine("Introdu n (n < 100000):");
          n = int.Parse(Console.ReadLine());
        }
      }
      catch (Exception ex)
      {
        Console.WriteLine("Eroare la citire: " + ex.Message);
        return;
      }

      if (n < 1 || n >= 100000)
      {
        Console.WriteLine("n trebuie să fie între 1 și 99999");
        return;
      }

      List result = new List();

      for (int i = 1; i <= n; i++)
      {
        int sum = DigitSum(i);

        if (sum % 2 == 1 && sum % 3 == 0)
        {
          result.Add(i);
        }
      }

      Console.WriteLine("Numere cu suma cifrelor impară și divizibilă cu 3:");
      if (result.Count == 0)
      {
        Console.WriteLine("Nu există numere cu această proprietate.");
      }
      else
      {
        foreach (int num in result)
        {
          Console.WriteLine($"{num} (suma cifrelor: {DigitSum(num)})");
        }
        Console.WriteLine($"\nTotal: {result.Count} numere");
      }

      {
        using (StreamWriter writer = new StreamWriter("output.txt"))

using System;
  using System.Collections.Generic;
  using System.Linq;

  class Punct
  {
    public double X { get; set; }
    public double Y { get; set; }

    public Punct(double x, double y)
    {
      X = x;
      Y = y;
    }

    public override string ToString() => $"({X}, {Y})";
  }

  class Program
  {
    static double CalculareArie(Punct p1, Punct p2, Punct p3)
    {
      return Math.Abs((p2.X - p1.X) * (p3.Y - p1.Y) - (p3.X - p1.X) * (p2.Y - p1.Y)) / 2.0;
    }

    static (Punct, Punct, Punct, double) GasireTriunghiMaxim(List puncte)
    {
      if (puncte.Count < 3)
        throw new ArgumentException("Sunt necesare cel puțin 3 puncte");

      Punct maxP1 = null, maxP2 = null, maxP3 = null;
      double arieMaxima = 0;

      for (int i = 0; i < puncte.Count - 2; i++)
      {
        for (int j = i + 1; j < puncte.Count - 1; j++)
        {
          for (int k = j + 1; k < puncte.Count; k++)
          {
            double arie = CalculareArie(puncte[i], puncte[j], puncte[k]);

            if (arie > arieMaxima)
            {
              arieMaxima = arie;
              maxP1 = puncte[i];
              maxP2 = puncte[j];
              maxP3 = puncte[k];
            }
          }
        }
      }

      return (maxP1, maxP2, maxP3, arieMaxima);
    }

    static void Main()
    {
      var puncte = new List();

      Console.Write("Nr-ul de puncte: ");
      if (!int.TryParse(Console.ReadLine(), out int numarPuncte) || numarPuncte < 3)
      {
        Console.WriteLine("Eroare: CEL PUTIN 3 PUNCTE");
        return;
      }

      for (int i = 0; i < numarPuncte; i++)
      {
        Console.WriteLine($"\nPunctul {i + 1}:");
        Console.Write("  Coordonata X: ");
        if (!double.TryParse(Console.ReadLine(), out double x))
        {
          Console.WriteLine("Eroare:NUMARUL NU ESTE VALID PT X.");
          i--;
          continue;
        }

        Console.Write("  Coordonata Y: ");
        if (!double.TryParse(Console.ReadLine(), out double y))
        {
          Console.WriteLine("Eroare:NUMARUL NU ESTE VALID PT Y.");
          i--;
          continue;
        }

        puncte.Add(new Punct(x, y));
      }

      Console.WriteLine("\n--- Punctele introduse ---");
      for (int i = 0; i < puncte.Count; i++)
        Console.WriteLine($"  Punctul {i + 1}: {puncte[i]}");

      var (p1, p2, p3, arieMaxima) = GasireTriunghiMaxim(puncte);
      Console.WriteLine("Triunghiul cu arie maximă:");
      Console.WriteLine($"  Punctele: {p1}, {p2}, {p3}");
      Console.WriteLine($"  Aria: {arieMaxima:F2}");
    }
  }

  class Depozit where T : Produs
  {
    private List produse;
    private string fisier;

    public Depozit(string numeFisier)
    {
      produse = new List();
      fisier = numeFisier;
      IncarcaProduse();
    }

    private void IncarcaProduse()
    {
      if (!File.Exists(fisier))
      {
        Console.WriteLine("Fișierul nu există. Se va crea o bază de date goală.");
        return;
      }

      try
      {
        using (StreamReader reader = new StreamReader(fisier))
        {
          string linie;
          while ((linie = reader.ReadLine()) != null)
          {
            if (string.IsNullOrWhiteSpace(linie)) continue;

            string[] parti = linie.Split(',');
            if (parti.Length < 8) continue;

            if (DateTime.TryParse(parti[6], out DateTime termen) &&
              decimal.TryParse(parti[3], out decimal pret) &&
              decimal.TryParse(parti[4], out decimal adaos) &&
              int.TryParse(parti[5], out int cantitate))
            {
              T produs = (T)Activator.CreateInstance(typeof(T),
                parti[0], parti[1], parti[2], pret, adaos, cantitate, termen, parti[7]);
              produse.Add(produs);
            }
          }
        }
        Console.WriteLine($"S-au încărcat {produse.Count} produse din fișier.");
      }
      catch (Exception ex)
      {
        Console.WriteLine($"Eroare la încărcarea fișierului: {ex.Message}");
      }
    }

    public void SalveazaProduse()
    {
      try
      {
        using (StreamWriter writer = new StreamWriter(fisier))
        {
          foreach (var produs in produse)
          {
            writer.WriteLine($"{produs.Cod},{produs.Furnizor},{produs.Unitatea},{produs.Pret},{produs.Adaos},{produs.Cantitate},{produs.TermenValabilitate:dd/MM/yyyy},{produs.Categoria}");
          }

import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.List;

public class Main {

    public static long countTrueWays(String line) {
        if (line == null) return 0;
        line = line.trim();
        if (line.isEmpty()) return 0;

        String[] tokens = line.split("\\s+");
        List<Boolean> vals = new ArrayList<>();
        List<String> ops = new ArrayList<>();

        for (int i = 0; i < tokens.length; i++) {
            String t = tokens[i].toLowerCase();
            if (i % 2 == 0) {
                if (t.equals("true")) vals.add(Boolean.TRUE);
                else if (t.equals("false")) vals.add(Boolean.FALSE);
                else throw new IllegalArgumentException("Expected boolean literal at token " + i + ": " + tokens[i]);
            } else {
                if (t.equals("and") || t.equals("or") || t.equals("xor")) ops.add(t);
                else throw new IllegalArgumentException("Expected operator at token " + i + ": " + tokens[i]);
            }
        }

        int n = vals.size();
        if (n == 0) return 0;

        long[][] t = new long[n][n];
        long[][] f = new long[n][n];

        for (int i = 0; i < n; i++) {
            t[i][i] = vals.get(i) ? 1 : 0;
            f[i][i] = vals.get(i) ? 0 : 1;
        }

        for (int len = 2; len <= n; len++) {
            for (int i = 0; i <= n - len; i++) {
                int j = i + len - 1;
                long trueWays = 0;
                long falseWays = 0;

                for (int k = i; k < j; k++) {
                    String op = ops.get(k);
                    long leftT = t[i][k];
                    long leftF = f[i][k];
                    long rightT = t[k + 1][j];
                    long rightF = f[k + 1][j];

                    long totalLeft = leftT + leftF;
                    long totalRight = rightT + rightF;

                    if (op.equals("and")) {
                        trueWays += leftT * rightT;
                        falseWays += totalLeft * totalRight - leftT * rightT;
                    } else if (op.equals("or")) {
                        falseWays += leftF * rightF;
                        trueWays += totalLeft * totalRight - leftF * rightF;
                    } else if (op.equals("xor")) {
                        trueWays += leftT * rightF + leftF * rightT;
                        falseWays += leftT * rightT + leftF * rightF;
                    }
                }

                t[i][j] = trueWays;
                f[i][j] = falseWays;
            }
        }

        return t[0][n - 1];
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String line = br.readLine();
        if (line == null) return;
        long ways;
        try {
            ways = countTrueWays(line);
        } catch (IllegalArgumentException ex) {
            System.err.println("Invalid expression: " + ex.getMessage());
            return;
        }
        System.out.println(ways);
    }
}

    used = [False] * len(word)
    current = []

    def backtrack() -> None:
        if len(current) == len(word):
            print("".join(current))
            return
        for index, letter in enumerate(word):

        public void Afisare()
        {
            if (prim == null)
            {
                Console.WriteLine("Lista este goala!");
                return;
            }
            Celula c = prim;
            while (c != null)
            {
                c.Afisare();
                c = c.urmatoare;
            }
        }

        public void Exclude()
        {
            if (prim == null)
            {
                Console.WriteLine("Lista este goala!");
                return;
            }
            Console.Write("Nume de sters: ");
            string numeCautat = Console.ReadLine();

            Celula c = prim, prev = null;
            while (c != null)
            {
                if (c.persoana.Nume == numeCautat)
                {
                    if (prev != null)
                        prev.urmatoare = c.urmatoare;
                    else
                        prim = c.urmatoare;
                    Console.WriteLine("Persoana stearsa!");
                    return;
                }
                prev = c;
                c = c.urmatoare;
            }
            Console.WriteLine("Persoana nu a fost gasita!");
        }

        public double Bani()
        {
            double total = 0;
            Celula c = prim;
            while (c != null)
            {
                total += c.CalculeazaCost();
                c = c.urmatoare;
            }
            return total;
        }

        public void Statistici()
        {
            int[] count = new int[11];
            double[] sume = new double[11];
            Celula c = prim;

            while (c != null)
            {
                int t = (int)c.persoana.GetTipCod();
                if (t >= 0 && t <= 10)
                {
                    count[t]++;
                    sume[t] += c.CalculeazaCost();
                }
                c = c.urmatoare;
            }

            Console.WriteLine();
            Console.WriteLine("=== STATISTICI ===");
            string[] nume = { "Persoana", "Student", "Angajat", "Student Angajat", "Administrator",
                            "Contabil", "Lector", "Sef Laborator", "Consultant Part-time",
                            "Jurist Part-time", "Student Bursier" };
            for (int i = 1; i <= 10; i++)
            {
                if (count[i] > 0)
                    Console.WriteLine($"{nume[i]}: {count[i]} persoane, Total: {sume[i]:F2} lei");
            }
        }

        public void SalvareInFisier(string fisier)
        {
            using (StreamWriter sw = new StreamWriter(fisier))
            {
                int n = 0;
                Celula c = prim;
                while (c != null)
                {
                    n++;
                    c = c.urmatoare;
                }
                sw.WriteLine(n);
                c = prim;
                while (c != null)
                {
                    c.persoana.ScriereInFisier(sw);
                    c = c.urmatoare;
                }
            }
            Console.WriteLine($"Date salvate in {fisier}");
        }

        public void CitireDinFisier(string fisier)
        {
            if (!File.Exists(fisier))
            {
                Console.WriteLine("Fisierul nu exista!");
                return;
            }
            using (StreamReader sr = new StreamReader(fisier))
            {
                int n = int.Parse(sr.ReadLine());
                for (int i = 0; i < n; i++)
                {
                    int tip = int.Parse(sr.ReadLine());
                    Persoana pers = null;
                    switch (tip)
                    {
                        case 1: pers = new Student("", "", 0, "", 0); break;
                        case 2: pers = new Angajat("", "", 0, 0, 0); break;
                        case 3: pers = new StudentAngajat("", "", 0, "", 0, 0, 0); break;
                        case 4: pers = new Administrator("", "", 0, 0, 0, 0, 0); break;
                        case 5: pers = new Contabil("", "", 0, 0, 0, 0, false); break;
                        case 6: pers = new Lector("", "", 0, 0, 0, 0, ""); break;
                        case 7: pers = new SefLaborator("", "", 0, 0, 0, 0, 0, "", 0); break;
                        case 8: pers = new ConsultantPartTime("", "", 0, 0, 0, 0, ""); break;
                        case 9: pers = new JuristPartTime("", "", 0, 0, 0, 0, "", 0, 0); break;
                        case 10: pers = new StudentBursier("", "", 0, "", 0, 0, 0); break;
                    }
                    if (pers != null)
                    {
                        pers.CitireDinFisier(sr);
                        Celula nou = new Celula();
                        nou.persoana = pers;
                        nou.urmatoare = prim;
                        prim = nou;
                    }
                }
            }
            Console.WriteLine($"Date incarcate din {fisier}");
        }
